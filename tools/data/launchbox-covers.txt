#
# make_launchbox_covers.py turns this into src/LaunchBoxCovers.h. The cover
# worker asks images.launchbox-app.com/<file> when Xbox Live sends nothing or
# its placeholder and xboxunity has no cover. Each line:
#
#   <title ID> = <LaunchBox image file>
#   name:<the Store's name for it> = <LaunchBox image file>
#

# (gamesdb.launchbox-app.com/Metadata.zip), matched by hand on 2026-11-12,
# North America or World first; the comment is LaunchBox's name and region.
