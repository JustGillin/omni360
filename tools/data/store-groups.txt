# Omni360 - Store discs grouped by hand
#
# make_store_titles.py groups a disc with a game of the same letter by name or
# title ID. These are regional releases under another name that neither
# catches, often under another letter. Each line:
#
#   <disc's zip, without .zip> = <title ID of the game it's a version of>
#
# The disc keeps its own name, shown as the edition on its version.

# "Matt Hazard no Gyakushuu" is the Japanese Eat Lead: The Return of Matt Hazard.

Scene It Ganz grosses Kino! (Germany) = 575207F5

# The Taiwanese disc of the World release, under its own title ID.
